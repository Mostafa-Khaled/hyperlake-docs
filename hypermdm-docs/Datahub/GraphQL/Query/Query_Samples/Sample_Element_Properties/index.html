Get Element Properties

Request​

Properties can be filtered by providing the keys argument. All properties will be returned by default.

{
  getDocument(id: 38) {
    ... on document_page {
      fullpath
      properties(keys: ["newsletter_confirm_mail", "myobject", "mylogo"]) {
        __typename
        ... on property_text {
          name
          text
        }
        ... on property_select {
          name
          text
        }
        ... on property_document {
          name
          type
          document {
            __typename
            ... on document_email {
              id
              fullpath
              controller
              action              
            }
          }
        }
        ... on property_asset {
          name
          asset {
            fullpath
            logothumb: fullpath(thumbnail: "content")
            filesize
          }
        }
        ... on property_object {
          name
          object {
            ... on object_Car {
              id
              name
              color
            }
          }
        }
      }
    }
  }
}

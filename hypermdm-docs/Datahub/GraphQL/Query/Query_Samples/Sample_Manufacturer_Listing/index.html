Get Manufacturer Listing

Request​

Manufacturer listing with limit 3 and offset 1, sorted by name.

{
  # 'first' is the limit
  # 'after' the offset
  getManufacturerListing(first: 3, after: 1, sortBy: "name") {
    edges {
      node {
        id
        name
        logo {
          id
          fullpath
        }
      }
    }
  }
}

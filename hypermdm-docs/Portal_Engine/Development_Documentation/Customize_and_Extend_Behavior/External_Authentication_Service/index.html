External Authentication Services

The portal engine provides several events to make it possible to integrate external authentication services via the login form.

  • HyperMDM\Bundle\PortalEngineBundle\Event\Auth\LoginCheckPasswordEvent
  • HyperMDM\Bundle\PortalEngineBundle\Event\Auth\LoginFieldTypeEvent
  • HyperMDM\Bundle\PortalEngineBundle\Event\Auth\LoginGetUserEvent
  • HyperMDM\Bundle\PortalEngineBundle\Event\Auth\LoginPasswordChangeableEvent

Important: It is needed to create a PortalUser data object for all external users too to make the portal engine work correctly.

Example​

This example event subscriber will allow HyperMDM admin users to login to the portal engine as admin users too.

<?php

namespace AppBundle\EventListener;

use HyperMDM\Bundle\PortalEngineBundle\Event\Auth\LoginCheckPasswordEvent;
use HyperMDM\Bundle\PortalEngineBundle\Event\Auth\LoginFieldTypeEvent;
use HyperMDM\Bundle\PortalEngineBundle\Event\Auth\LoginGetUserEvent;
use HyperMDM\Bundle\PortalEngineBundle\Event\Auth\LoginPasswordChangeableEvent;
use HyperMDM\Model\DataObject\PortalUser;
use HyperMDM\Model\DataObject\Service;
use HyperMDM\Model\User;
use HyperMDM\Tool\Authentication;
use Symfony\Component\EventDispatcher\EventSubscriberInterface;

class PortalEngineAuthenticationSubscriber implements EventSubscriberInterface
{
    const USER_FOLDER = '/Portal Engine/Users/HyperMDM';

    public static function getSubscribedEvents()
    {
        return [
            LoginFieldTypeEvent::class => 'onLoginFieldType',
            LoginGetUserEvent::class => 'onLoginGetUser',
            LoginCheckPasswordEvent::class => 'onLoginCheckPassword',
            LoginPasswordChangeableEvent::class => 'onPasswordChangeable',
        ];
    }

    public function onLoginFieldType(LoginFieldTypeEvent $event)
    {
        //switch the login form to allow non-email user names
        $event->setUseEmailField(false);
    }

    public function onLoginGetUser(LoginGetUserEvent $event)
    {
        $portalUser = null;
        $hypermdmUser = User::getByName($event->getUserName());
        if ($hypermdmUser && $hypermdmUser->getAdmin()) {
            $portalUser = $this->getOrCreatePortalUserByHyperMDMUser($hypermdmUser);
            $event->setPortalUser($portalUser);
        }
    }

    public function onLoginCheckPassword(LoginCheckPasswordEvent $event)
    {
        $portalUser = $event->getPortalUser();
        if ($portalUser->getExternalUserId()) {
            $hypermdmUser = User::getByName($portalUser->getExternalUserId());
            $success = Authentication::isValidUser($hypermdmUser)
                && Authentication::verifyPassword(
                    $hypermdmUser,
                    $event->getPassword()
                );

            if ($success) {
                $event->setLoginValid(true);
            }
        }
    }

    public function onPasswordChangeable(LoginPasswordChangeableEvent $event)
    {
        /* If an external authentication service is used it might be needed to disable
         * the recover/change password functionality.
         *
         * In this example this is not necessary.
         */

        //$event->setPasswordChangeable(false);
    }

    protected function getOrCreatePortalUserByHyperMDMUser(User $hypermdmUser)
    {
        // to make the portal engine work it is needed to create a portal user data object
        $portalUser = PortalUser::getByExternalUserId($hypermdmUser->getName(), 1);
        if (empty($portalUser)) {
            $portalUser = new PortalUser();
            $portalUser->setHyperMDMUser($hypermdmUser->getId())
                ->setExternalUserId($hypermdmUser->getName())
                ->setAdmin(true)
                ->setKey($hypermdmUser->getName())
                ->setEmail($hypermdmUser->getEmail())
                ->setFirstname($hypermdmUser->getFirstname())
                ->setLastname($hypermdmUser->getLastname())
                ->setParent(Service::createFolderByPath(self::USER_FOLDER))
                ->setUseHyperMDMUserPassword(true)
                ->setPublished(true)
                ->setKey(Service::getUniqueKey($portalUser))
                ->setPreferredLanguage($hypermdmUser->getLanguage());

            $portalUser->save();
        }

        return $portalUser;
    }
}

# add this to your container service definition
services:
    AppBundle\EventListener\PortalEngineAuthenticationSubscriber:
        tags:
            - { name: kernel.event_subscriber }
